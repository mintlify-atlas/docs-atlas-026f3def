This guide walks you through adding WorkOS authentication to a React Router 7+ application using `@workos-inc/authkit-react-router`. By the end you will have a working callback route, sign-in and sign-up redirects, and a protected dashboard route that requires a valid session.

<Steps>

  <Step title="Install the package">

    Add `@workos-inc/authkit-react-router` to your project:

    <CodeGroup>

    ```bash npm
    npm install @workos-inc/authkit-react-router
    ```

    ```bash yarn
    yarn add @workos-inc/authkit-react-router
    ```

    ```bash pnpm
    pnpm add @workos-inc/authkit-react-router
    ```

    </CodeGroup>

  </Step>

  <Step title="Configure environment variables">

    Create or update `.env.local` at the root of your project with your WorkOS credentials:

    ```bash .env.local
    WORKOS_CLIENT_ID="client_..."
    WORKOS_REDIRECT_URI="http://localhost:5173/callback"
    WORKOS_COOKIE_PASSWORD="<your-32-character-password>"
    WORKOS_API_KEY="sk_test_..."  # optional — omit for keyless (public-client) mode
    ```

    <Note>
      `WORKOS_COOKIE_PASSWORD` is used to encrypt the session cookie with iron-session. It must be **at least 32 characters** long. You can generate a strong password with `openssl rand -base64 32`.
    </Note>

    <Tip>
      Retrieve your **Client ID**, **API Key**, and configure your **Redirect URI** in the [WorkOS Dashboard](https://dashboard.workos.com). The redirect URI you set here must match exactly what you register under **Redirects** in your WorkOS application settings.
    </Tip>

  </Step>

  <Step title="Add the callback route">

    AuthKit needs a dedicated route to handle the OAuth callback from WorkOS. Create `app/routes/callback.ts` and export `authLoader()` as the route loader:

    ```ts app/routes/callback.ts
    import { authLoader } from '@workos-inc/authkit-react-router';

    export const loader = authLoader();
    ```

    `authLoader` verifies the PKCE + CSRF state, exchanges the authorization code for tokens, encrypts them into a session cookie, and redirects the user back to your app.

    <Note>
      The path of this route must match the value of `WORKOS_REDIRECT_URI`. For example, if your redirect URI is `http://localhost:5173/callback`, the file should be `app/routes/callback.ts` so that React Router maps it to `/callback`.
    </Note>

    You can also control where the user lands after a successful sign-in:

    ```ts app/routes/callback.ts
    import { authLoader } from '@workos-inc/authkit-react-router';

    export const loader = authLoader({ returnPathname: '/dashboard' });
    ```

  </Step>

  <Step title="Create sign-in and sign-up routes">

    `getSignInUrl` and `getSignUpUrl` each return a `{ url, headers }` object. The `headers` contain a short-lived `Set-Cookie` that binds the PKCE verifier and CSRF nonce to the current browser. **These headers must travel to the browser on the same redirect that sends the user to AuthKit** — create dedicated redirect-only routes rather than pre-rendering the URL into a link.

    ```ts app/routes/login.ts
    import { redirect, type LoaderFunctionArgs } from 'react-router';
    import { getSignInUrl } from '@workos-inc/authkit-react-router';

    export async function loader({ request }: LoaderFunctionArgs) {
      const url = new URL(request.url);
      const { url: authUrl, headers } = await getSignInUrl(
        url.searchParams.get('returnTo') ?? undefined,
        request,
      );
      return redirect(authUrl, { headers });
    }
    ```

    ```ts app/routes/signup.ts
    import { redirect, type LoaderFunctionArgs } from 'react-router';
    import { getSignUpUrl } from '@workos-inc/authkit-react-router';

    export async function loader({ request }: LoaderFunctionArgs) {
      const url = new URL(request.url);
      const { url: authUrl, headers } = await getSignUpUrl(
        url.searchParams.get('returnTo') ?? undefined,
        request,
      );
      return redirect(authUrl, { headers });
    }
    ```

    From any page in your app, link to these routes:

    ```tsx
    import { Link } from 'react-router';

    export default function HomePage() {
      return (
        <>
          <Link to="/login">Log in</Link>
          <Link to="/signup">Sign up</Link>
        </>
      );
    }
    ```

    Passing `request` as the second argument to `getSignInUrl` / `getSignUpUrl` ensures the `Secure` attribute on the PKCE cookie matches your app's live protocol — important in local development where your app runs on `http://localhost` even if `WORKOS_REDIRECT_URI` points to an `https://` URL.

  </Step>

  <Step title="Protect a route">

    Use `authkitLoader` with `ensureSignedIn: true` on any route that requires authentication. Unauthenticated visitors are automatically redirected to WorkOS AuthKit. The loader merges `user`, `sessionId`, `organizationId`, `role`, `permissions`, `entitlements`, `featureFlags`, and `impersonator` into the data returned by `useLoaderData`.

    ```tsx app/routes/dashboard.tsx
    import { type LoaderFunctionArgs, useLoaderData } from 'react-router';
    import { authkitLoader } from '@workos-inc/authkit-react-router';

    export const loader = (args: LoaderFunctionArgs) =>
      authkitLoader(args, { ensureSignedIn: true });

    export default function Dashboard() {
      const { user } = useLoaderData<typeof loader>();

      return (
        <main>
          <h1>Welcome back{user.firstName ? `, ${user.firstName}` : ''}!</h1>
          <p>Email: {user.email}</p>
        </main>
      );
    }
    ```

    Because `ensureSignedIn: true` is set, TypeScript knows `user` is never `null` inside this component. For routes where both signed-in and anonymous visitors are allowed, omit the option and check `user !== null` yourself.

  </Step>

</Steps>
